def bump : ContDiffBump (0:ℝ) :=
  {rIn:=1,rOut:=2,rIn_pos:=by norm_num,rIn_lt_rOut:=by norm_num}
def observable (x : ℝ) : ℝ := bump.normed (volume : Measure ℝ) x * ‖x‖^(3/2:ℝ)

theorem real_noncompact_seven_and_actual_C1_domain :
    (¬ Integrable (fun _ : ℝ => (7:ℝ)) volume) ∧
    ContDiff ℝ 1 observable ∧ HasCompactSupport observable ∧
    ∃ μ : Measure ℝ, IsProbabilityMeasure μ ∧
      ∃ D : Lp ℝ 2 μ →ₗ.[ℝ] Lp ℝ 2 μ,
        Dense (D.domain : Set (Lp ℝ 2 μ)) ∧ D.IsClosable ∧
        (∀ (a H : Lp ℝ 2 μ), (a,H) ∈ D.graph ↔
          ∃ φ : ℝ → ℝ, ContDiff ℝ ∞ φ ∧ HasCompactSupport φ ∧
            a =ᵐ[μ] φ ∧ H =ᵐ[μ] gradient φ) ∧
        (∃ u : D.closure.domain,
          (u : Lp ℝ 2 μ) =ᵐ[μ] (fun _ => (7:ℝ)) ∧ D.closure u=0) ∧
        (∃ hp : MemLp observable 2 μ, ∃ hq : MemLp (gradient observable) 2 μ,
          (hp.toLp _,hq.toLp _) ∈ D.closure.graph) ∧
        ∀ u : D.closure.domain, D.closure u=0 ↔
          ∃ c : ℝ, (u : Lp ℝ 2 μ) =ᵐ[μ] (fun _ => c) := by
  have hn : ¬ Integrable (fun _ : ℝ => (7:ℝ)) volume := by
    intro hi
    rcases integrable_const_iff.mp hi with hz | hfin
    · norm_num at hz
    · letI := hfin
      have hm := measure_ne_top (volume : Measure ℝ) Set.univ
      rw [Real.volume_univ] at hm
      exact hm rfl
  have ho : ContDiff ℝ 1 observable :=
    (contDiff_infty.mp bump.contDiff_normed 1).mul
      (contDiff_norm_rpow (by norm_num : (1:ℝ)<3/2))
  have hc : HasCompactSupport observable := bump.hasCompactSupport_normed.mul_right
  have hs := Tests.WeightedC1GradientDomain.actual_nonquadratic_score_original_domain
  dsimp only at hs
  obtain ⟨R,S,hR,hS,hcond,href,hy⟩ := hs
  let μ := S 0
  haveI : IsProbabilityMeasure μ := hS.isProbabilityMeasure 0
  obtain ⟨hdensity,hW,hI,hZ,D,hd,hD,hclosed,hgraph,hconstants,hkernel,hscore⟩ := hy 0
  have hgrad : Continuous (gradient observable) :=
    (toDual ℝ ℝ).symm.continuous.comp (ho.continuous_fderiv one_ne_zero)
  have hgc : HasCompactSupport (gradient observable) := by
    apply HasCompactSupport.of_support_subset_isCompact hc.isCompact
    intro x hx
    by_contra hn
    exact hx (by simp [gradient,fderiv_of_notMem_tsupport ℝ hn])
  have hp : MemLp observable 2 μ := ho.continuous.memLp_of_hasCompactSupport hc
  have hq : MemLp (gradient observable) 2 μ := hgrad.memLp_of_hasCompactSupport hgc
  refine ⟨hn,ho,hc,μ,inferInstance,D,hd,hD,hgraph,hconstants 7,⟨hp,hq,?_⟩,hkernel⟩
  exact AutoSamplingTheory.TechnicalLemmas.FunctionalInequalities.WeightedC1GradientDomain.c1_in_closed_gradient
    μ D hD hgraph ho hp hq
#print axioms real_noncompact_seven_and_actual_C1_domain
