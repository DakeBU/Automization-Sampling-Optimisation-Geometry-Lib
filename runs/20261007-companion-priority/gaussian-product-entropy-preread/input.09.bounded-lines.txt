5: # Product-measure coordinate replacement leaves
6: 
7: Small `Measure.pi` facts for replacing one coordinate by an independent sample.
8: These are the product-measure and Fubini leaves needed by tensorization and
9: coordinate-slice arguments.  They deliberately do not introduce conditional
10: expectations, entropy, LSI, kernels, or weak-Fokker--Planck statements.
11: -/

25: /-- The coordinate-replacement map `(y, x) ↦ Function.update x i y` is measurable. -/
26: theorem measurable_update_prod_pi (i : Fin n) :
27:     Measurable (fun p : Ω × (Fin n → Ω) => Function.update p.2 i p.1) := by
28:   have h : (fun p : Ω × (Fin n → Ω) => Function.update p.2 i p.1) =
29:       (fun q : (Fin n → Ω) × Ω => Function.update q.1 i q.2) ∘ Prod.swap := rfl
30:   rw [h]
31:   exact (measurable_update' (a := i)).comp measurable_swap

33: /-- Replacing one coordinate of a product sample by an independent sample from
34: that coordinate preserves the product law. -/
35: theorem map_update_prod_pi (i : Fin n) :
36:     Measure.map (fun p : Ω × (Fin n → Ω) => Function.update p.2 i p.1)
37:       ((μs i).prod (Measure.pi μs)) = Measure.pi μs := by
38:   symm
39:   apply Measure.pi_eq
40:   intro s hs
41:   rw [Measure.map_apply]
42:   · have preimage_eq :
43:         (fun p : Ω × (Fin n → Ω) => Function.update p.2 i p.1) ⁻¹' (Set.univ.pi s) =
44:           (s i) ×ˢ (Set.univ.pi (fun j => if j = i then Set.univ else s j)) := by
45:       ext ⟨y, x⟩
46:       simp only [Set.mem_preimage, Set.mem_pi, Set.mem_univ, true_implies, Set.mem_prod]
47:       constructor
48:       · intro h
49:         constructor
50:         · have := h i
51:           simp only [Function.update_self] at this
52:           exact this
53:         · intro j
54:           by_cases hj : j = i
55:           · simp [hj]
56:           · have := h j
57:             simp only [Function.update_of_ne hj] at this
58:             simp [hj, this]
59:       · intro ⟨hy, hx⟩ j
60:         by_cases hj : j = i
61:         · subst hj
62:           simp only [Function.update_self]
63:           exact hy
64:         · simp only [Function.update_of_ne hj]
65:           have := hx j
66:           simp only [hj, ↓reduceIte] at this
67:           exact this
68:     rw [preimage_eq, Measure.prod_prod]
69:     have pi_eq :
70:         Measure.pi μs (Set.univ.pi (fun j => if j = i then Set.univ else s j)) =
71:           ∏ j : Fin n, (if j = i then 1 else μs j (s j)) := by
72:       rw [Measure.pi_pi]
73:       congr 1 with j
74:       by_cases hj : j = i
75:       · subst hj
76:         simp only [↓reduceIte, measure_univ]
77:       · simp only [hj, ↓reduceIte]
78:     rw [pi_eq]
79:     have h_ite : ∀ j, (if j = i then (1 : ℝ≥0∞) else μs j (s j)) =
80:         (if j ∈ Finset.univ.erase i then μs j (s j) else 1) := by
81:       intro j
82:       by_cases hj : j = i
83:       · simp [hj]
84:       · simp [hj]
85:     simp_rw [h_ite]
86:     rw [Fintype.prod_extend_by_one, mul_comm,
87:       Finset.prod_erase_mul _ _ (Finset.mem_univ i)]
88:   · exact measurable_update_prod_pi i
89:   · exact MeasurableSet.univ_pi (fun j => hs j)

91: /-- Measure-preserving wrapper for coordinate replacement under a product law. -/
92: theorem measurePreserving_update_prod_pi (i : Fin n) :
93:     MeasurePreserving (fun p : Ω × (Fin n → Ω) => Function.update p.2 i p.1)
94:       ((μs i).prod (Measure.pi μs)) (Measure.pi μs) :=
95:   ⟨measurable_update_prod_pi i, map_update_prod_pi i⟩

97: /-- Averaging a function after one-coordinate replacement over the fresh
98: coordinate and the original product sample recovers its product-law integral. -/
99: theorem integral_update_prod_pi_eq_integral
100:     {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [CompleteSpace E]
101:     (i : Fin n) {f : (Fin n → Ω) → E}
102:     (hf : Integrable f (Measure.pi μs)) :
103:     ∫ y, ∫ x, f (Function.update x i y) ∂(Measure.pi μs) ∂(μs i) =
104:       ∫ z, f z ∂(Measure.pi μs) := by
105:   set g : Ω × (Fin n → Ω) → E := fun p => f (Function.update p.2 i p.1) with hg_def
106:   have h1 :
107:       ∫ y, ∫ x, f (Function.update x i y) ∂(Measure.pi μs) ∂(μs i) =
108:         ∫ p, g p ∂((μs i).prod (Measure.pi μs)) := by
109:     rw [integral_prod]
110:     exact (measurePreserving_update_prod_pi (μs := μs) i).integrable_comp
111:       hf.aestronglyMeasurable |>.mpr hf
112:   rw [h1, hg_def, ← integral_map]
113:   · rw [map_update_prod_pi (μs := μs) i]
114:   · exact (measurable_update_prod_pi i).aemeasurable
115:   · rw [map_update_prod_pi (μs := μs) i]
116:     exact hf.aestronglyMeasurable
