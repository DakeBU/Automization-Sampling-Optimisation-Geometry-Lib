-- The zero-dimensional residual factor is the unique probability unit.
example : stdGaussian (EuclideanSpace ℝ (Fin 0)) = Measure.dirac 0 := by
  apply Measure.ext
  intro s hs
  by_cases h : (0 : EuclideanSpace ℝ (Fin 0)) ∈ s
  · have heq : s = Set.univ := by
      ext x
      simpa only [Subsingleton.elim x 0, Set.mem_univ, iff_true] using h
    simp [heq]
  · have heq : s = ∅ := by
      ext x
      simpa only [Subsingleton.elim x 0, Set.mem_empty_iff_false, iff_false] using h
    simp [heq]
