def Q (λ x : ℝ) : ℝ := λ/2*x^2+7
theorem Q_curvature (λ z a : ℝ) :
    fderiv ℝ (fderiv ℝ (Q λ)) z a a = λ*‖a‖^2 := by
  have hd (w : ℝ) : HasDerivAt (Q λ) (λ*w) w := by
    convert (((hasDerivAt_id w).pow 2).const_mul (λ/2)).add_const 7 using 1 <;>
      first | rfl | (simp [Q]; ring)
  have hD : fderiv ℝ (Q λ) = fun w => (λ*w) • ContinuousLinearMap.id ℝ ℝ := by
    funext w
    rw [fderiv_eq_smul_deriv, (hd w).deriv]
  have hdd : HasDerivAt (fun w : ℝ => (λ*w) • ContinuousLinearMap.id ℝ ℝ)
      (λ • ContinuousLinearMap.id ℝ ℝ) z :=
    ((hasDerivAt_id z).const_mul λ).smul_const _
  rw [hD, fderiv_eq_smul_deriv, hdd.deriv]
  simp only [ContinuousLinearMap.smul_apply, ContinuousLinearMap.id_apply,
    smul_eq_mul, Real.norm_eq_abs, sq_abs]
  ring

-- lambda=1/2, kappa=2, eta=1: the source LOWER Hessian constant is sharp.
-- Genuine translated posterior, all y/a, mass1/L2 and both exact constants.
theorem sharp_lower_endpoint :
    let μ := (volume : Measure ℝ).tilted (fun x => -Q (1/2) x)
    let C := ((Real.sqrt (2*Real.pi*(1:ℝ)))⁻¹)^Module.finrank ℝ ℝ
    let Vη := fun y : ℝ => -Real.log
      (C*∫ x, Real.exp (-Q (1/2) x-‖y-x‖^2/(2*(1:ℝ))) ∂(volume : Measure ℝ))
    let R := fun y : ℝ => μ.tilted (fun x => -‖x-y‖^2/(2*(1:ℝ)))
    IsProbabilityMeasure μ ∧ ContDiff ℝ 2 Vη ∧
      (∀ y, IsProbabilityMeasure (R y) ∧ MemLp id 2 (R y) ∧
        ∀ a, covarianceBilin (R y) a a=(2/3:ℝ)*‖a‖^2 ∧
          fderiv ℝ (fderiv ℝ Vη) y a a=(1/3:ℝ)*‖a‖^2) ∧
      LipschitzWith (1:ℝ≥0) (gradient Vη) := by
  have hV : ContDiff ℝ 2 (Q (1/2)) := by unfold Q; fun_prop
  have hh : ∀ z a : ℝ, (2:ℝ)⁻¹*‖a‖^2 ≤
      fderiv ℝ (fderiv ℝ (Q (1/2))) z a a ∧
      fderiv ℝ (fderiv ℝ (Q (1/2))) z a a ≤ ‖a‖^2 := by
    intro z a
    rw [Q_curvature]
    constructor <;> nlinarith [sq_nonneg ‖a‖]
  have heq : ∀ z a : ℝ, ((1/2:ℝ≥0):ℝ)*‖a‖^2 ≤
      fderiv ℝ (fderiv ℝ (Q (1/2))) z a a ∧
      fderiv ℝ (fderiv ℝ (Q (1/2))) z a a ≤ ((1/2:ℝ≥0):ℝ)*‖a‖^2 := by
    intro z a
    rw [Q_curvature]
    norm_num
  have h := AutoSamplingTheory.ExampleCases.SmoothedPicardHMC.SmoothedHessianBounds.smoothed_hessian_bounds
    (κ:=2) (η:=1) (by norm_num) hV hh (by norm_num) (by norm_num)
  have old := AutoSamplingTheory.ExampleCases.SmoothedPicardHMC.SmoothedHessianUpper.smoothed_hessian_upper
    (α:=1/2) (β:=1/2) (η:=1) (by norm_num) le_rfl hV heq (by norm_num)
  refine ⟨h.1,h.2.1,?_,h.2.2.2⟩
  intro y
  refine ⟨(h.2.2.1 y).1,(h.2.2.1 y).2.1,?_⟩
  intro a
  have cu := (h.2.2.1 y).2.2.1 a
  have cl := old.1 y a
  have hu := old.2 y a
  have hl := ((h.2.2.1 y).2.2.2 a).1
  norm_num at cu cl hu hl
  constructor <;> apply le_antisymm <;> nlinarith

-- Exact coefficient at a=2 rules out accidental direction-independent covariance.
theorem sharp_direction_two :
    let μ := (volume : Measure ℝ).tilted (fun x => -Q (1/2) x)
    let R := fun y : ℝ => μ.tilted (fun x => -‖x-y‖^2/(2*(1:ℝ)))
    ∀ y, covarianceBilin (R y) (2:ℝ) (2:ℝ)=8/3 := by
  intro y
  have h := ((sharp_lower_endpoint.2.2.1 y).2.2 (2:ℝ)).1
  norm_num at h ⊢
  exact h

-- Additive source partition convention is preserved for the actual nonunit
-- exponential offset7; do not identify U and V_eta as equal by fiat.
theorem true_source_constant_shift :
    let μ := (volume : Measure ℝ).tilted (fun x => -Q (1/2) x)
    let C := ((Real.sqrt (2*Real.pi*(1:ℝ)))⁻¹)^Module.finrank ℝ ℝ
    let A := fun y : ℝ => C*∫ x, Real.exp
      (-Q (1/2) x-‖y-x‖^2/(2*(1:ℝ))) ∂(volume : Measure ℝ)
    (∀ y, 0<A y) ∧ (∫ y, A y)=(∫ x, Real.exp (-Q (1/2) x)) ∧
      ∀ y, -Real.log (C*∫ x, Real.exp (-‖y-x‖^2/(2*(1:ℝ))) ∂μ) =
        -Real.log (A y)+Real.log (∫ x, Real.exp (-Q (1/2) x)) := by
  have hV : ContDiff ℝ 2 (Q (1/2)) := by unfold Q; fun_prop
  have hl : ∀ z a : ℝ, (1/2:ℝ)*‖a‖^2 ≤ fderiv ℝ (fderiv ℝ (Q (1/2))) z a a := by
    intro z a
    rw [Q_curvature]
  have h := AutoSamplingTheory.ExampleCases.SmoothedPicardHMC.SmoothedGibbsPotential.smoothed_gibbs_potential
    (α:=1/2) (η:=1) (by norm_num) hV hl (by norm_num)
  exact ⟨h.2.2.2.2.2.2.1,h.2.2.2.2.1,h.2.2.2.2.2.2.2.2⟩
