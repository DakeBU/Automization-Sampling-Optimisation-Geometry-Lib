def V (x : ℝ) : ℝ := (3/8)*x^2 + (1/8)*Real.sin x
theorem V_deriv (x : ℝ) : HasDerivAt V ((3/4)*x+(1/8)*Real.cos x) x := by
  convert (((hasDerivAt_id x).pow 2).const_mul (3/8)).add
    ((Real.hasDerivAt_sin x).const_mul (1/8)) using 1 <;>
    first | rfl | (simp [V,Pi.add_apply,mul_comm];ring)
theorem V_gradient (x : ℝ) : gradient V x = (3/4)*x+(1/8)*Real.cos x := by
  exact (V_deriv x).hasGradientAt.gradient
theorem V_curvature (x v : ℝ) :
    (2:ℝ)⁻¹*‖v‖^2 ≤ fderiv ℝ (fderiv ℝ V) x v v ∧
    fderiv ℝ (fderiv ℝ V) x v v ≤ ‖v‖^2 := by
  have hd : fderiv ℝ V = fun x => ((3/4)*x+(1/8)*Real.cos x) • ContinuousLinearMap.id ℝ ℝ := by
    funext x
    ext
    rw [fderiv_eq_smul_deriv,(V_deriv x).deriv]
    simp
  have hd2 : HasDerivAt (fun w : ℝ => (3/4)*w+(1/8)*Real.cos w)
      ((3/4)-(1/8)*Real.sin x) x := by
    convert ((hasDerivAt_id x).const_mul (3/4)).add
      ((Real.hasDerivAt_cos x).const_mul (1/8)) using 1 <;>
      (try dsimp only [id]) <;> first | rfl | ring
  rw [hd,fderiv_eq_smul_deriv,(hd2.smul_const (ContinuousLinearMap.id ℝ ℝ)).deriv]
  have hs := Real.sin_le_one x
  have hns := Real.neg_one_le_sin x
  simp only [ContinuousLinearMap.smul_apply,ContinuousLinearMap.id_apply,smul_eq_mul,
    Real.norm_eq_abs,sq_abs]
  constructor <;> norm_num <;> nlinarith [sq_nonneg v]

def eta (s : ℝ) : ℝ := if s=0 then 1 else 1/2
theorem eta_measurable : Measurable eta := by
  exact measurable_const.ite (measurableSet_singleton 0) measurable_const
theorem eta_pos (s : ℝ) : 0<eta s := by
  unfold eta;split_ifs <;> norm_num
theorem eta_small (s : ℝ) : eta s≤1 := by
  unfold eta;split_ifs <;> norm_num

-- Genuine nonsymmetric nonquadratic potential, variable step, real centers.
-- At eta1 and center0 the true proximal point is nonzero, so a zero selector fails.
theorem nonquadratic_actual_variable_eta :
    ∃ (p : ℝ → ℝ) (K : Kernel ℝ ℝ), Measurable p ∧
      p 0≠0 ∧
      (∀ s, p s+eta s*gradient V (p s)=s) ∧
      (∀ z, V (p 0)+(1/2)*‖p 0‖^2+(3/4)*‖z-p 0‖^2
          ≤ V z+(1/2)*‖z‖^2) ∧
      (∀ s t, eta s=eta t → ‖p s-p t‖≤‖s-t‖) ∧
      Measurable (fun q : ℝ × ℝ => gradient V (p q.1+Real.sqrt (eta q.1)*q.2)) ∧
      (∀ s t z w, eta s=eta t →
        ‖gradient V (p s+Real.sqrt (eta s)*z)-gradient V (p t+Real.sqrt (eta t)*w)‖
          ≤ ‖s-t‖+Real.sqrt (eta s)*‖z-w‖) ∧
      IsMarkovKernel K ∧
      (∀ s, K s=(stdGaussian ℝ).map (fun z => gradient V (p s+Real.sqrt (eta s)*z))) := by
  have hv : ContDiff ℝ 2 V := by unfold V;fun_prop
  obtain ⟨p,hp,heq,hmin,hnexp,hG,hGL,K,hK,hKs⟩ :=
    full_range_proximal_gaussian_oracle (κ:=2) (by norm_num) hv V_curvature
      eta_measurable (y:=id) measurable_id eta_pos eta_small
  have hp0 : p 0≠0 := by
    intro h
    have h0 := heq 0
    rw [h,V_gradient] at h0
    norm_num [eta] at h0
  refine ⟨p,K,hp,hp0,?_,?_,?_,?_,?_,hK,?_⟩
  · intro s;simpa only [smul_eq_mul,id_eq] using heq s
  · intro z
    have h := (hmin 0 z).1
    norm_num [eta] at h
    simpa only [sub_zero,Real.norm_eq_abs,sq_abs] using h
  · simpa only [id_eq] using hnexp
  · simpa only [smul_eq_mul] using hG
  · simpa only [smul_eq_mul,id_eq] using hGL
  · simpa only [smul_eq_mul] using hKs

-- Actual Gaussian-output law with varying eta and fixed center1:
-- the two literal minimizers differ, so input-only nonexpansivity cannot cross eta.
theorem different_eta_guard_is_needed :
    ∃ K : Kernel ℝ ℝ, IsMarkovKernel K ∧
      K 0=(stdGaussian ℝ).map (fun z => (1/2:ℝ)+z) ∧
      K 1=(stdGaussian ℝ).map (fun z => (2/3:ℝ)+Real.sqrt (1/2:ℝ)*z) ∧
      (∫ w, ‖w-(1/2:ℝ)‖^2 ∂K 0)=1 ∧
      (∫ w, ‖w-(2/3:ℝ)‖^2 ∂K 1)=1/2 ∧
      ¬(‖(1/2:ℝ)-(2/3:ℝ)‖≤‖(1:ℝ)-1‖) := by
  obtain ⟨K,hK,h⟩ := Tests.FullRangeProximalGaussianOracle.quadratic_variable_eta
    eta_measurable (y:=fun _ : ℝ => 1) measurable_const eta_pos eta_small
  have h0 := h 0
  have h1 := h 1
  norm_num [eta] at h0 h1
  refine ⟨K,hK,h0.1,?_,?_,?_,?_⟩
  · simpa only [show Real.sqrt (1/2:ℝ)=(Real.sqrt 2)⁻¹ by norm_num] using h1.1
  · simpa only [Real.norm_eq_abs,sq_abs] using h0.2.1
  · simpa only [Real.norm_eq_abs,sq_abs] using h1.2.1
  · norm_num [Real.norm_eq_abs]
