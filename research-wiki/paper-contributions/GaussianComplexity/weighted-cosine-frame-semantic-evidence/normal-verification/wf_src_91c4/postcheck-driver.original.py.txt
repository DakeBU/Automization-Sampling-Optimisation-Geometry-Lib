def sha(b): return hashlib.sha256(b).hexdigest()
def canon(x): return json.dumps(x, ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode()
def digest_file(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''): h.update(block)
    return h.hexdigest()
def tracked_entry(name):
    path = ROOT / name
    info = path.lstat()
    if stat.S_ISLNK(info.st_mode): return name, {'kind': 'symlink', 'target': os.readlink(path), 'mode': stat.S_IMODE(info.st_mode)}
    if stat.S_ISREG(info.st_mode): return name, {'kind': 'regular', 'bytes': info.st_size, 'mode': stat.S_IMODE(info.st_mode), 'sha256': digest_file(path)}
    return name, {'kind': 'other', 'mode': info.st_mode}
def save(path, value): path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
