-- Positive nonlinear growth exercises Fernique, whereas exp(-3*norm) alone is bounded.
theorem nonlinear_positive_parameter :
    Integrable (fun x : ℝ => ‖x‖) (stdGaussian ℝ) ∧
      Integrable (fun x : ℝ => Real.exp (3*(‖x‖-∫ z : ℝ, ‖z‖ ∂stdGaussian ℝ)))
        (stdGaussian ℝ) := by
  have hf : LipschitzWith 1 (fun x : ℝ => ‖x‖) := by
    apply LipschitzWith.of_dist_le_mul
    intro x y
    simpa only [dist_eq_norm, Real.norm_eq_abs, NNReal.coe_one, one_mul] using
      abs_norm_sub_norm_le x y
  have h := integrable_and_integrable_exp_centered_of_lipschitz (μ := stdGaussian ℝ) hf
  exact ⟨h.1, h.2 3⟩

abbrev E0 := EuclideanSpace ℝ (Fin 0)
-- This extracts the new domain conjuncts from the actual source producer, not just dirac existence.
theorem actual_zero_dimension_domains :
    ∃ K : Kernel Unit E0, IsMarkovKernel K ∧ K () = Measure.dirac (0:E0) ∧
      Integrable (fun w : E0 => w) (K ()) ∧
        ∀ (a : E0) (t : ℝ), Integrable
          (fun w => Real.exp (t * inner ℝ a (w-∫ v, v ∂K ()))) (K ()) := by
  have hH : ∀ x v : E0, (1:ℝ)⁻¹*‖v‖^2 ≤
      fderiv ℝ (fderiv ℝ (fun _ : E0 => (7:ℝ))) x v v ∧
      fderiv ℝ (fderiv ℝ (fun _ : E0 => (7:ℝ))) x v v ≤ ‖v‖^2 := by
    intro x v
    have hv : v=0 := Subsingleton.elim _ _
    simp [hv]
  obtain ⟨p,_hp,_heq,_hmin,_hnexp,_hG,_hLip,K,hK,hKs,hDomain⟩ :=
    full_range_proximal_gaussian_oracle (κ:=1) (by norm_num)
      contDiff_const hH (eta:=fun _ : Unit => (4:ℝ)) (y:=fun _ => (0:E0))
      measurable_const measurable_const (by norm_num)
  have hf : (fun z => gradient (fun _ : E0 => (7:ℝ)) (p ()+Real.sqrt (4:ℝ) • z))=
      fun _ : E0 => (0:E0) := by
    funext z
    exact Subsingleton.elim _ _
  refine ⟨K,hK,?_,(hDomain ()).1,(hDomain ()).2⟩
  rw [hKs (),hf]
  simp
