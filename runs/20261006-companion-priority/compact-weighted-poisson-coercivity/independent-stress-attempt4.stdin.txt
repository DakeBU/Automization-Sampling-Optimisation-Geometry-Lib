def bump : ContDiffBump (0:ℝ) := {rIn:=1,rOut:=2,rIn_pos:=by norm_num,rIn_lt_rOut:=by norm_num}
def chi (x:ℝ) := bump.normed (volume:Measure ℝ) x * (x^2)^(5/2:ℝ)
def W : ℝ → ℝ → ℝ := fun y z =>
  (3/8)*((1/2:ℝ)*(y+z))^2+(1/8)*Real.sin ((1/2:ℝ)*(y+z))+‖z-y‖^2/(8*(1/2:ℝ))
def WeightedResult {E:Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E]
    [FiniteDimensional ℝ E] [MeasurableSpace E] [BorelSpace E]
    (μ:Measure E) (D:Lp ℝ 2 μ →ₗ.[ℝ] Lp E 2 μ)
    (m:ℝ) (v A:E→ℝ) (G:E→E) : Prop :=
  MemLp v 2 μ ∧ MemLp G 2 μ ∧ MemLp A 2 μ ∧
  ∃ vW AW:Lp ℝ 2 μ, ∃ GW:Lp E 2 μ,
    vW =ᵐ[μ] v ∧ AW =ᵐ[μ] A ∧ GW =ᵐ[μ] G ∧
    (vW,GW) ∈ D.closure.graph ∧ m*‖GW‖^2≤‖AW‖^2

theorem actual_nonquadratic_half_weighted_domain_coercivity :
  ∃ S:Kernel ℝ ℝ, IsMarkovKernel S ∧
    ∀ y, ∃ D:Lp ℝ 2 (S y) →ₗ.[ℝ] Lp ℝ 2 (S y), D.IsClosable ∧
    ∀ ε:ℝ,0<ε → ∀ f:Lp ℝ 2 (S y), ∃ u:D.closure.domain,
      let G:=fun x=>chi x • D.closure u x+(u:Lp ℝ 2 (S y)) x • gradient chi x
      let F:=fun x=>chi x*(ε*(u:Lp ℝ 2 (S y)) x+
        inner ℝ (gradient (W y) x) (D.closure u x)-f x)+
        2*inner ℝ (gradient chi x) (D.closure u x)+(u:Lp ℝ 2 (S y)) x*Δ chi x
      WeightedResult (S y) D (5/8:ℝ)
        (fun x=>chi x*(u:Lp ℝ 2 (S y)) x)
        (fun x=>inner ℝ (gradient (W y) x) (G x)-F x) G := by
  obtain ⟨R,S,hR,hS,hcond,hSR,hfiber⟩ :=
    Tests.CompactWeightedPoissonCoercivity.actual_c2_cutoff_weighted_coercivity
      (η:=1/2) (by norm_num) (by norm_num)
  refine ⟨S,hS,?_⟩
  intro y
  obtain ⟨hSy,hW,hI,hZ,D,hDense,hClose,hClosed,hgraph,hsolve⟩ := hfiber y
  refine ⟨D,hClose,?_⟩
  intro ε hε f
  obtain ⟨u,hu,huL,hGL,hder,hr2,hlocal,hvm,hgm,hfm,hvs,hgs,hfs,hg,hp,hw⟩ := hsolve ε hε f
  refine ⟨u,?_⟩
  change WeightedResult (S y) D (((1/2:ℝ)+1/(1/2:ℝ))/4) _ _ _ at hw
  have hconst : (((1/2:ℝ)+1/(1/2:ℝ))/4) = (5/8:ℝ) := by norm_num
  rw [hconst] at hw
  exact hw

abbrev E0 := EuclideanSpace ℝ (Fin 0)
theorem actual_zero_dim_constant7_weighted (m:ℝ) :
    let μ := (volume:Measure E0).tilted (fun _=>-(7:ℝ))
    μ Set.univ=1 ∧
    ∃ D:Lp ℝ 2 μ →ₗ.[ℝ] Lp E0 2 μ, D.IsClosable ∧
      ∀ ε:ℝ,0<ε → ∀ f:Lp ℝ 2 μ, ∃ u:D.closure.domain,
        WeightedResult μ D m (fun x=>(2:ℝ)*(u:Lp ℝ 2 μ) x)
          (fun x=> -(2:ℝ)*(ε*(u:Lp ℝ 2 μ) x-f x))
          (fun _=> (0:E0)) := by
  have hI:Integrable (fun _:E0=>Real.exp (-(7:ℝ))) := integrable_const _
  have hZ : 0<∫ _:E0,Real.exp (-(7:ℝ)) := integral_exp_pos hI
  let μ := (volume:Measure E0).tilted (fun _=>-(7:ℝ))
  let : IsProbabilityMeasure μ := isProbabilityMeasure_tilted hI
  refine ⟨measure_univ,?_⟩
  obtain ⟨D,hDense,hD,hClosed,hgraph⟩ :=
    AutoSamplingTheory.TechnicalLemmas.FunctionalInequalities.WeightedGradient.compact_gradient_closable
      (fun _:E0=>(7:ℝ)) contDiff_const hI
  refine ⟨D,hD,?_⟩
  intro ε hε f
  obtain ⟨u,hu,huL,hGL,hd,hr2,hl,hp,hcut⟩ :=
    AutoSamplingTheory.TechnicalLemmas.FunctionalInequalities.LocalizedWeakResolvent.localized_weak_resolvent
      (fun _:E0=>(7:ℝ)) contDiff_const hI D hD hgraph ε hε f
  have hc:HasCompactSupport (fun _:E0=>(2:ℝ)) :=
    HasCompactSupport.of_support_subset_isCompact isCompact_univ (Set.subset_univ _)
  obtain ⟨hv,hg,hf,hvs,hgs,hfs,hfirst,hsecond⟩ := hcut (fun _=>(2:ℝ)) contDiff_const hc
  have hcurv : ∀ x a:E0,m*‖a‖^2≤fderiv ℝ (fderiv ℝ (fun _:E0=>(7:ℝ))) x a a := by
    intro x a
    have ha : a=0 := Subsingleton.elim _ _
    simp [ha]
  have hw :=
    AutoSamplingTheory.TechnicalLemmas.FunctionalInequalities.CompactWeightedPoissonCoercivity.compact_weak_poisson_weighted_coercivity
      (fun _:E0=>(7:ℝ)) contDiff_const hI hZ m hcurv μ rfl D hD hgraph
      _ _ _ (tsupport (fun _:E0=>(2:ℝ))) hc hvs hfs hgs hv hf hg hfirst hsecond
  refine ⟨u,?_⟩
  change WeightedResult μ D m _ _ _ at hw
  have hzero (x:E0) : D.closure u x=0 := Subsingleton.elim _ _
  simpa only [gradient_fun_const, InnerProductSpace.laplacian_const, hzero, smul_zero,
    add_zero, inner_zero_left, inner_zero_right, zero_mul, mul_zero, sub_zero,
    zero_sub, zero_add, neg_mul, Pi.zero_apply] using hw
