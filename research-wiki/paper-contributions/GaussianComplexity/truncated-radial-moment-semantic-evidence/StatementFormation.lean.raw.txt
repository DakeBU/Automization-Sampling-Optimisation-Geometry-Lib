import AutoSamplingTheory.TechnicalLemmas.Probability.StdGaussianRadialExponential

open MeasureTheory ProbabilityTheory
open scoped ENNReal
set_option autoImplicit false

def PlannedFirstMoment : Prop := ∀ {d q : ℕ} (hq : 2 * q + 1 < d),
  let E := EuclideanSpace ℝ (Fin (q + 1))
  let μ : Measure E := stdGaussian E
  let R : E → ℝ := fun x => ‖x‖ ^ 2
  let L : ℝ := 2 * ((q + 1 : ℕ) : ℝ) + 3 * Real.log (1000 * (d : ℝ))
  let badMoment : E → ℝ := Set.indicator {x : E | L < R x} R
  let tail : ℝ → ℝ := fun s => (μ {x : E | L + s < R x}).toReal
  Integrable badMoment μ ∧ IntegrableOn tail (Set.Ioi (0 : ℝ)) ∧
    (∫ x, badMoment x ∂μ) = L * tail 0 + (∫ s in Set.Ioi (0 : ℝ), tail s) ∧
    (∫ x, badMoment x ∂μ) ≤ (L + 3) / (1000 * (d : ℝ))

def PlannedBiasConsumer : Prop := ∀ {d q : ℕ} (hq : 2 * q + 1 < d),
  let E := EuclideanSpace ℝ (Fin (q + 1))
  let μ : Measure E := stdGaussian E
  let r : ℝ := ((q + 1 : ℕ) : ℝ)
  let R : E → ℝ := fun x => ‖x‖ ^ 2
  let L : ℝ := 2 * r + 3 * Real.log (1000 * (d : ℝ))
  let a : ℝ := (∫ x, Set.indicator {y : E | R y ≤ L} R x ∂μ) / r
  0 ≤ a ∧ a ≤ 1 ∧
    1 - a = (∫ x, Set.indicator {y : E | L < R y} R x ∂μ) / r ∧
    1 - a ≤ (L + 3) / (1000 * r * (d : ℝ))

#check PlannedFirstMoment
#check PlannedBiasConsumer
