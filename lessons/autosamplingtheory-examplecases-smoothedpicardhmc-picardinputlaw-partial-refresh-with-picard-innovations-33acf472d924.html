    Mathematical statement

    On the nested product of the incoming law, a fresh standard Gaussian, and a finite product of standard Gaussians, the refreshed X and P0 and every Picard innovation G_j are measurable and square-integrable with their exact consumer-ready moment bounds.

    \[B_{\rm refresh}=M+(1-e^{-h})d,\qquad \mathbb E\|G_j^{[0]}\|^2=d.\]

    Proof roadmap

    1. 1Build the probability space

      The nested product measure represents freshness of the first refresh Gaussian and the first Picard innovation array.

    2. 2Reuse the refresh calculation

      The previous theorem already proves measurability, integrability, and the corrected OU moment budget; product integration lifts it unchanged over the extra innovation array.

    3. 3Read each Gaussian coordinate

      Evaluation at j is measurable. Finite-product marginal integration returns the original standard Gaussian law, so every coordinate has exactly dimension d as its squared-norm moment.

    4. 4Feed the real variables to the next theorem

      The focused test passes the product-law coordinate projections to the two-layer Picard moment theorem, and separately checks a nondegenerate one-dimensional Gaussian coordinate moment.

    Full assumptions and notation

    All objects and hypotheses

    • The incoming phase-state law is a probability measure with an integrable combined square bounded by M≥0.
    • The phase length is nonnegative and the Picard index type is finite.
    • E is a complete finite-dimensional real inner-product Borel space.

    Mathematical proof

    1. Build the probability space

    The nested product measure represents freshness of the first refresh Gaussian and the first Picard innovation array.

    \[\mu=(\nu\otimes\gamma_d)\otimes\bigotimes_{j\in[J]}\gamma_d.\]
    Corresponding Lean step

    Measure.prod constructs the refresh factor and Measure.pi constructs the finite innovation array.

    2. Reuse the refresh calculation

    The previous theorem already proves measurability, integrability, and the corrected OU moment budget; product integration lifts it unchanged over the extra innovation array.

    \[\mathbb E_\mu[\|X-x_*\|^2+\|P_0\|^2]\le B_{\rm refresh}.\]
    Corresponding Lean step

    Integrable.comp_fst and integral_prod transport the established result.

    3. Read each Gaussian coordinate

    Evaluation at j is measurable. Finite-product marginal integration returns the original standard Gaussian law, so every coordinate has exactly dimension d as its squared-norm moment.

    \[G_j(w)=w_{\rm array}(j),\quad \mathbb E\|G_j\|^2=d.\]
    Corresponding Lean step

    integrable_comp_eval and integral_comp_eval are the pinned Mathlib APIs.

    4. Feed the real variables to the next theorem

    The focused test passes the product-law coordinate projections to the two-layer Picard moment theorem, and separately checks a nondegenerate one-dimensional Gaussian coordinate moment.

    \[(X,P_0,(G_j)_j)\longrightarrow\text{PicardCenterMoment}.\]
    Corresponding Lean step

    Tests.SmoothedPicardHMCPartialMomentumRefresh checks symbolic API composition in dimension zero and the exact positive-dimensional Gaussian moment in E=R.

    theorem partial_refresh_with_picard_innovations
    (ν : Measure (E × E)) [IsProbabilityMeasure ν]
    (xstar : E) {h M : ℝ} (hh : 0 ≤ h) (hM : 0 ≤ M)
    (hstateI : Integrable (fun s : E × E => ‖s.1 - xstar‖ ^ 2 + ‖s.2‖ ^ 2) ν)
    (hstate : (∫ s : E × E, ‖s.1 - xstar‖ ^ 2 + ‖s.2‖ ^ 2 ∂ν) ≤ M) :
    let a := Real.exp (-h / 2)
    let sigma := Real.sqrt (1 - Real.exp (-h))
    let μRefresh := ν.prod (stdGaussian E)
    let γι := Measure.pi (fun _ : ι => stdGaussian E)
    let μ := μRefresh.prod γι
    let X : (((E × E) × E) × (ι → E)) → E := fun w => w.1.1.1
    let P0 : (((E × E) × E) × (ι → E)) → E :=
      fun w => a • w.1.1.2 + sigma • w.1.2
    let G : ι → (((E × E) × E) × (ι → E)) → E := fun j w => w.2 j
    let BRefresh := M + (1 - Real.exp (-h)) * (Module.finrank ℝ E : ℝ)
    0 ≤ BRefresh ∧
      Measurable X ∧ Measurable P0 ∧
      Integrable (fun w => ‖X w - xstar‖ ^ 2 + ‖P0 w‖ ^ 2) μ ∧
      (∫ w, ‖X w - xstar‖ ^ 2 + ‖P0 w‖ ^ 2 ∂μ) ≤ BRefresh ∧
      (∀ j, Measurable (G j)) ∧
      (∀ j, Integrable (fun w => ‖G j w‖ ^ 2) μ) ∧
      (∀ j, (∫ w, ‖G j w‖ ^ 2 ∂μ) = (Module.finrank ℝ E : ℝ))

    theorem partial_refresh_with_picard_innovations
    (ν : Measure (E × E)) [IsProbabilityMeasure ν]
    (xstar : E) {h M : ℝ} (hh : 0 ≤ h) (hM : 0 ≤ M)
    (hstateI : Integrable (fun s : E × E => ‖s.1 - xstar‖ ^ 2 + ‖s.2‖ ^ 2) ν)
    (hstate : (∫ s : E × E, ‖s.1 - xstar‖ ^ 2 + ‖s.2‖ ^ 2 ∂ν) ≤ M) :
    let a := Real.exp (-h / 2)
    let sigma := Real.sqrt (1 - Real.exp (-h))
    let μRefresh := ν.prod (stdGaussian E)
    let γι := Measure.pi (fun _ : ι => stdGaussian E)
    let μ := μRefresh.prod γι
    let X : (((E × E) × E) × (ι → E)) → E := fun w => w.1.1.1
    let P0 : (((E × E) × E) × (ι → E)) → E :=
      fun w => a • w.1.1.2 + sigma • w.1.2
    let G : ι → (((E × E) × E) × (ι → E)) → E := fun j w => w.2 j
    let BRefresh := M + (1 - Real.exp (-h)) * (Module.finrank ℝ E : ℝ)
    0 ≤ BRefresh ∧
      Measurable X ∧ Measurable P0 ∧
      Integrable (fun w => ‖X w - xstar‖ ^ 2 + ‖P0 w‖ ^ 2) μ ∧
      (∫ w, ‖X w - xstar‖ ^ 2 + ‖P0 w‖ ^ 2 ∂μ) ≤ BRefresh ∧
      (∀ j, Measurable (G j)) ∧
      (∀ j, Integrable (fun w => ‖G j w‖ ^ 2) μ) ∧
      (∀ j, (∫ w, ‖G j w‖ ^ 2 ∂μ) = (Module.finrank ℝ E : ℝ)) := by
  classical
  dsimp only
  obtain ⟨hX, hP0, hrefreshI, hrefreshEq, hrefresh⟩ :=
    PartialMomentumRefresh.partial_momentum_refresh_second_moment
      ν xstar hh hstateI hstate
  let μRefresh := ν.prod (stdGaussian E)
  let γι := Measure.pi (fun _ : ι => stdGaussian E)
  have hvar : 0 ≤ 1 - Real.exp (-h) :=
    sub_nonneg.mpr (Real.exp_le_one_iff.mpr (by linarith))
  have hB : 0 ≤ M + (1 - Real.exp (-h)) * (Module.finrank ℝ E : ℝ) :=
    add_nonneg hM (mul_nonneg hvar (Nat.cast_nonneg _))
  have hrefreshOuter : Integrable
      (fun w : ((E × E) × E) × (ι → E) =>
        ‖w.1.1.1 - xstar‖ ^ 2 +
          ‖Real.exp (-h / 2) • w.1.1.2 +
            Real.sqrt (1 - Real.exp (-h)) • w.1.2‖ ^ 2)
      (μRefresh.prod γι) := hrefreshI.comp_fst γι
  have hrefreshOuterInt :
      (∫ w : ((E × E) × E) × (ι → E),
          ‖w.1.1.1 - xstar‖ ^ 2 +
            ‖Real.exp (-h / 2) • w.1.1.2 +
              Real.sqrt (1 - Real.exp (-h)) • w.1.2‖ ^ 2
          ∂μRefresh.prod γι) ≤
        M + (1 - Real.exp (-h)) * (Module.finrank ℝ E : ℝ) := by
    rw [integral_prod _ hrefreshOuter]
    simpa [γι, μRefresh] using hrefresh
  obtain ⟨hgaussI, hgauss⟩ :=
    integrable_norm_sq_and_integral_stdGaussian (E := E)
  have hcoordI (j : ι) : Integrable (fun z : ι → E => ‖z j‖ ^ 2) γι :=
    by
      simpa [γι] using
        (integrable_comp_eval (X := fun _ : ι => E)
          (μ := fun _ : ι => stdGaussian E) (i := j) hgaussI)
  have hcoordOuterI (j : ι) : Integrable
      (fun w : ((E × E) × E) × (ι → E) => ‖w.2 j‖ ^ 2)
      (μRefresh.prod γι) := (hcoordI j).comp_snd μRefresh
  have hcoordOuter (j : ι) :
      (∫ w : ((E × E) × E) × (ι → E), ‖w.2 j‖ ^ 2
          ∂μRefresh.prod γι) = (Module.finrank ℝ E : ℝ) := by
    rw [integral_prod _ (hcoordOuterI j)]
    simp [μRefresh, γι,
      integral_comp_eval (X := fun _ : ι => E) (μ := fun _ : ι => stdGaussian E)
        (i := j) hgaussI.aestronglyMeasurable,
      hgauss]
  exact ⟨hB, by fun_prop, by fun_prop, hrefreshOuter, hrefreshOuterInt,
    fun _ => by fun_prop, hcoordOuterI, hcoordOuter⟩
