52: /-- **Lebesgue dominated convergence theorem** provides sufficient conditions under which almost
53:   everywhere convergence of a sequence of functions implies the convergence of their integrals.
54:   We could weaken the condition `bound_integrable` to require `HasFiniteIntegral bound μ` instead
55:   (i.e. not requiring that `bound` is measurable), but in all applications proving integrability
56:   is easier. -/
57: theorem tendsto_integral_of_dominated_convergence {F : ℕ → α → G} {f : α → G} (bound : α → ℝ)
58:     (F_measurable : ∀ n, AEStronglyMeasurable (F n) μ) (bound_integrable : Integrable bound μ)
59:     (h_bound : ∀ n, ∀ᵐ a ∂μ, ‖F n a‖ ≤ bound a)
60:     (h_lim : ∀ᵐ a ∂μ, Tendsto (fun n => F n a) atTop (𝓝 (f a))) :
61:     Tendsto (fun n => ∫ a, F n a ∂μ) atTop (𝓝 <| ∫ a, f a ∂μ) := by
62:   simp only [integral_eq_setToFun]
63:   exact tendsto_setToFun_of_dominated_convergence (dominatedFinMeasAdditive_weightedSMul μ)
64:     bound F_measurable bound_integrable h_bound h_lim
65: 
66: /-- Lebesgue dominated convergence theorem for filters with a countable basis -/
67: theorem tendsto_integral_filter_of_dominated_convergence {ι} {l : Filter ι} [l.IsCountablyGenerated]
68:     {F : ι → α → G} {f : α → G} (bound : α → ℝ) (hF_meas : ∀ᶠ n in l, AEStronglyMeasurable (F n) μ)
69:     (h_bound : ∀ᶠ n in l, ∀ᵐ a ∂μ, ‖F n a‖ ≤ bound a) (bound_integrable : Integrable bound μ)
70:     (h_lim : ∀ᵐ a ∂μ, Tendsto (fun n => F n a) l (𝓝 (f a))) :
71:     Tendsto (fun n => ∫ a, F n a ∂μ) l (𝓝 <| ∫ a, f a ∂μ) := by
72:   simp only [integral_eq_setToFun]
73:   exact tendsto_setToFun_filter_of_dominated_convergence (dominatedFinMeasAdditive_weightedSMul μ)
74:     bound hF_meas h_bound bound_integrable h_lim
