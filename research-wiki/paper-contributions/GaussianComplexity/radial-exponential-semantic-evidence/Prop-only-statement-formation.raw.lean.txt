import Mathlib.Probability.Distributions.Gaussian.Multivariate
import Mathlib.Analysis.SpecialFunctions.Pow.Real
open MeasureTheory ProbabilityTheory
set_option autoImplicit false
def PlannedProducer : Prop := ∀ {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E]
    [FiniteDimensional ℝ E] [MeasurableSpace E] [BorelSpace E],
  Integrable (fun x : E => Real.exp (‖x‖ ^ 2 / 3)) (stdGaussian E) ∧
    (∫ x : E, Real.exp (‖x‖ ^ 2 / 3) ∂stdGaussian E) =
      (3 : ℝ) ^ ((Module.finrank ℝ E : ℝ) / 2)

def PlannedConsumer : Prop := ∀ {d q : ℕ} (hq : 2 * q + 1 < d),
  let E := EuclideanSpace ℝ (Fin (q + 1))
  let μ : Measure E := stdGaussian E
  let L : ℝ := 2 * ((q + 1 : ℕ) : ℝ) + 3 * Real.log (1000 * (d : ℝ))
  μ {x : E | L < ‖x‖ ^ 2} ≤ ENNReal.ofReal ((1000 * (d : ℝ))⁻¹)

#check PlannedProducer
#check PlannedConsumer
