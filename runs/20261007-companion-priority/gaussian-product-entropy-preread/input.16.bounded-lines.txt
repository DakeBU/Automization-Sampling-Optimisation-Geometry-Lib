33: lemma self_sub_one_lt_mul_log {x : ℝ} (h0 : 0 ≤ x) (h1 : x ≠ 1) : x - 1 < x * x.log := by
34:   by_cases hx_pos : 0 < x
35:   · nlinarith [Real.log_inv x, Real.log_lt_sub_one_of_pos (inv_pos.mpr hx_pos) (by aesop),
36:       mul_inv_cancel₀ hx_pos.ne']
37:   · cases lt_or_eq_of_le h0 <;> aesop
38: 
39: lemma self_sub_one_le_mul_log {x : ℝ} (h0 : 0 ≤ x) : x - 1 ≤ x * x.log := by
40:   rcases eq_or_ne x 1 with rfl | h1
41:   · simp
42:   · exact le_of_lt (self_sub_one_lt_mul_log h0 h1)

44: @[fun_prop]
45: lemma continuous_mul_log : Continuous fun x ↦ x * log x := by
46:   rw [continuous_iff_continuousAt]
47:   intro x
48:   obtain hx | rfl := ne_or_eq x 0
49:   · exact (continuous_id'.continuousAt).mul (continuousAt_log hx)
50:   rw [ContinuousAt, zero_mul]
51:   simp_rw [mul_comm _ (log _)]
52:   nth_rewrite 1 [← nhdsWithin_univ]
53:   have : (Set.univ : Set ℝ) = Set.Iio 0 ∪ Set.Ioi 0 ∪ {0} := by ext; simp [em]
54:   rw [this, nhdsWithin_union, nhdsWithin_union]
55:   simp only [nhdsWithin_singleton, Filter.tendsto_sup]
56:   refine ⟨⟨tendsto_log_mul_self_nhdsLT_zero, ?_⟩, ?_⟩
57:   · simpa only [rpow_one] using tendsto_log_mul_rpow_nhdsGT_zero zero_lt_one
58:   · convert! tendsto_pure_nhds (fun x ↦ log x * x) 0
59:     simp

61: @[fun_prop]
62: lemma Continuous.mul_log {α : Type*} [TopologicalSpace α] {f : α → ℝ} (hf : Continuous f) :
63:     Continuous fun a ↦ f a * log (f a) := continuous_mul_log.comp hf
