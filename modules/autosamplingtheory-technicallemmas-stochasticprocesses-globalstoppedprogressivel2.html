    (eta : GlobalLocalProgressiveL2Integrand filtration mu)
    (k : ℕ) :
    AEStronglyMeasurable
      (processFunction (globalStoppedIntegrand hUsual eta k))
      (processTimeMeasure mu (dyadicHorizon k)) := by
  let extSwap : Omega × ℝ≥0 → ℝ := fun z =>
