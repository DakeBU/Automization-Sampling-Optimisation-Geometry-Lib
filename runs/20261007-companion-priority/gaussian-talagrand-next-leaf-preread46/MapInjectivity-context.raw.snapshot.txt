open scoped RealInnerProductSpace

variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E]

/-- Hilbert-space monotonicity of a point map. -/
def IsMonotoneMap (T : E → E) : Prop :=
  ∀ x y, 0 ≤ ⟪T x - T y, x - y⟫
